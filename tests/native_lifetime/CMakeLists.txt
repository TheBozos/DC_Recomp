cmake_minimum_required(VERSION 3.21)
project(DCNativeLifetime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BOZO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BOZO_APP OFF CACHE BOOL "" FORCE)
include(CTest)
add_subdirectory("${DC_ROOT}/BozoEngine" engine)
if(BUILD_TESTING)
  add_subdirectory("${DC_ROOT}/BozoEngine/tests" engine_tests)
  add_executable(dc-model-lifetime-tests "${DC_ROOT}/tests/dc_model_lifetime_tests.cpp" "${DC_ROOT}/runtime/dc_native.cpp")
  target_link_libraries(dc-model-lifetime-tests PRIVATE bozo_host)
  target_compile_definitions(dc-model-lifetime-tests PRIVATE DC_DEFAULT_TEXTURES="")
  if(MINGW)
    target_link_options(dc-model-lifetime-tests PRIVATE -static)
  endif()
  add_test(NAME dc_native_model_lifetime COMMAND dc-model-lifetime-tests)
endif()
