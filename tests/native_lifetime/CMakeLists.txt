cmake_minimum_required(VERSION 3.21)
project(DCNativeLifetime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BOZO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BOZO_APP OFF CACHE BOOL "" FORCE)
include(CTest)
add_subdirectory("${DC_ROOT}/BozoEngine" engine)
if(BUILD_TESTING)
  add_subdirectory("${DC_ROOT}/BozoEngine/tests" engine_tests)
  add_executable(dc-model-lifetime-tests "${DC_ROOT}/tests/dc_model_lifetime_tests.cpp" "${DC_ROOT}/runtime/dc_native.cpp")
  target_link_libraries(dc-model-lifetime-tests PRIVATE bozo_host)
  target_compile_definitions(dc-model-lifetime-tests PRIVATE DC_DEFAULT_TEXTURES="")
  if(MINGW)
    target_link_options(dc-model-lifetime-tests PRIVATE -static)
  endif()
  add_test(NAME dc_native_model_lifetime COMMAND dc-model-lifetime-tests)
  add_executable(dc-fishing-line-tests "${DC_ROOT}/tests/dc_fishing_line_tests.cpp"
    "${DC_ROOT}/runtime/dc_native.cpp" "${DC_ROOT}/runtime/dc_port.cpp" "${DC_ROOT}/runtime/dc_sound.cpp")
  target_link_libraries(dc-fishing-line-tests PRIVATE bozo_host)
  target_compile_definitions(dc-fishing-line-tests PRIVATE DC_DEFAULT_TEXTURES="")
  if(MINGW)
    target_link_options(dc-fishing-line-tests PRIVATE -static)
  endif()
  add_test(NAME dc_native_fishing_line COMMAND dc-fishing-line-tests)
  add_executable(dc-debug-font-tests "${DC_ROOT}/tests/dc_debug_font_tests.cpp"
    "${DC_ROOT}/runtime/dc_native.cpp" "${DC_ROOT}/runtime/dc_port.cpp" "${DC_ROOT}/runtime/dc_sound.cpp")
  target_link_libraries(dc-debug-font-tests PRIVATE bozo_host)
  target_compile_definitions(dc-debug-font-tests PRIVATE DC_DEFAULT_TEXTURES="")
  if(MINGW)
    target_link_options(dc-debug-font-tests PRIVATE -static)
  endif()
  add_test(NAME dc_native_debug_font COMMAND dc-debug-font-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(dc-packet-audit-tests "${DC_ROOT}/tests/dc_packet_audit_tests.cpp"
    "${DC_ROOT}/runtime/dc_native.cpp" "${DC_ROOT}/runtime/dc_port.cpp" "${DC_ROOT}/runtime/dc_sound.cpp")
  target_link_libraries(dc-packet-audit-tests PRIVATE bozo_host)
  target_compile_definitions(dc-packet-audit-tests PRIVATE DC_DEFAULT_TEXTURES="")
  if(MINGW)
    target_link_options(dc-packet-audit-tests PRIVATE -static)
  endif()
  add_test(NAME dc_native_packet_guards COMMAND ${Python3_EXECUTABLE} -I "${DC_ROOT}/tools/dc_packet_audit.py"
    --root "${DC_ROOT}" --registry $<TARGET_FILE:dc-packet-audit-tests>)
  add_executable(dc-indirect-facts "${DC_ROOT}/tools/dc_indirect_facts.cpp")
  target_link_libraries(dc-indirect-facts PRIVATE bozo_recomp_lib)
  add_executable(dc-indirect-trace-tests "${DC_ROOT}/tests/dc_indirect_trace_tests.cpp" "${DC_ROOT}/runtime/dc_indirect_trace.cpp")
  target_link_libraries(dc-indirect-trace-tests PRIVATE bozo_runtime_headers)
  target_include_directories(dc-indirect-trace-tests PRIVATE "${DC_ROOT}/runtime")
  if(MINGW)
    target_link_options(dc-indirect-facts PRIVATE -static)
    target_link_options(dc-indirect-trace-tests PRIVATE -static)
  endif()
  add_test(NAME dc_indirect_tools COMMAND ${Python3_EXECUTABLE} -I "${DC_ROOT}/tests/test_indirect_audit.py"
    $<TARGET_FILE:dc-indirect-facts> $<TARGET_FILE:dc-indirect-trace-tests>)
  add_test(NAME dc_pointer_alias_tools COMMAND ${Python3_EXECUTABLE} -I "${DC_ROOT}/tests/test_alias_audit.py")
  add_test(NAME dc_callee_effect_tools COMMAND ${Python3_EXECUTABLE} -I "${DC_ROOT}/tests/test_callee_audit.py")
  add_test(NAME dc_loadchara_bounds_tools COMMAND ${Python3_EXECUTABLE} -I "${DC_ROOT}/tests/test_loadchara_audit.py")
  add_test(NAME dc_readinfo_effect_tools COMMAND ${Python3_EXECUTABLE} -I "${DC_ROOT}/tests/test_readinfo_audit.py")
