foreach(image dc title dun)
  set(chunks 32)
  if(NOT image STREQUAL "dc")
    set(chunks 4)
  endif()
  set(files ${dc_gen}/${image}_functions.h ${dc_gen}/${image}_table.cpp)
  math(EXPR last "${chunks} - 1")
  foreach(i RANGE ${last})
    string(LENGTH "${i}" len)
    if(len EQUAL 1)
      set(n "00${i}")
    else()
      set(n "0${i}")
    endif()
    list(APPEND files ${dc_gen}/${image}_${n}.cpp)
  endforeach()
  # Patches belong to an image's compiled table. TITLE and DUN reuse virtual
  # addresses, so their adaptations must never become global runtime overrides.
  set(native_adaptation_inputs ${CMAKE_SOURCE_DIR}/recomp/${image}_native.patches)
  set(native_adaptations --patches ${native_adaptation_inputs})
  set(image_externals ${dc_overlay_externals})
  if(image STREQUAL "dc")
    set(image_externals ${dc_externals})
  endif()
  list(APPEND files ${dc_gen}/${image}.indirect.json)
  set(indirect_instrumentation)
  if(DC_AUDIT_INDIRECT)
    set(indirect_instrumentation COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_SOURCE_DIR}/tools/dc_instrument_indirect.py ${dc_gen} ${image})
  endif()
  add_custom_command(OUTPUT ${files}
    COMMAND bozo-recomp ${dc_gen}/${image}.elf ${dc_gen} --prefix ${image} --symbol-prefix ${image}_ --files ${chunks} --no-comments --externals ${image_externals} ${native_adaptations}
    COMMAND dc-indirect-facts ${dc_gen}/${image}.elf ${native_adaptation_inputs} ${dc_gen}/${image}.indirect.json
    ${indirect_instrumentation}
    DEPENDS bozo-recomp dc-indirect-facts ${dc_gen}/${image}.elf ${image_externals} ${native_adaptation_inputs}
            tools/dc_instrument_indirect.py)
  list(APPEND dc_sources ${files})
endforeach()

  function(dc_packet_wrapper_fixture kind)
    set(fixture_dir ${CMAKE_BINARY_DIR}/${kind}_reference)
    set(outputs)
    foreach(variant ${kind}_ref ${kind}_native)
      list(APPEND outputs ${fixture_dir}/${variant}_000.cpp ${fixture_dir}/${variant}_table.cpp
                          ${fixture_dir}/${variant}_functions.h)
    endforeach()
    add_custom_command(OUTPUT ${outputs}
      COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_SOURCE_DIR}/tests/make_packet_wrapper_fixture.py ${kind}
        ${DC_GAME_DIR}/SCUS_971.11 ${CMAKE_SOURCE_DIR}/recomp/dc_native.patches ${fixture_dir}/reference.elf
      COMMAND bozo-recomp ${fixture_dir}/reference.elf ${fixture_dir} --prefix ${kind}_ref
        --symbol-prefix ${kind}_ref_ --files 1 --no-comments --externals ${fixture_dir}/reference.externals
        --patches ${fixture_dir}/reference.baseline.patches
      COMMAND bozo-recomp ${fixture_dir}/reference.elf ${fixture_dir} --prefix ${kind}_native
        --symbol-prefix ${kind}_native_ --files 1 --no-comments --externals ${fixture_dir}/reference.native.externals
        --patches ${fixture_dir}/reference.patches
      DEPENDS bozo-recomp tests/make_packet_wrapper_fixture.py ${DC_GAME_DIR}/SCUS_971.11 recomp/dc_native.patches)
    set(wrapper_test tests/dc_${kind}_wrapper_tests.cpp)
    if(kind STREQUAL "texanim_state" OR kind STREQUAL "item_state")
      set(wrapper_test tests/dc_copy_state_wrapper_tests.cpp)
    endif()
    if(kind STREQUAL "shadow_fast" OR kind STREQUAL "shadow_fast2" OR kind STREQUAL "shadow")
      set(wrapper_test tests/dc_shadow_wrapper_tests.cpp)
    endif()
    if(kind MATCHES "^(packet_init|begin_frame|resident_chain|loading_init)$")
      set(wrapper_test tests/dc_resident_packet_tests.cpp)
    endif()
    if(kind MATCHES "^(packet_reserve|buffer_clear|packet_handle|packet_get)$")
      set(wrapper_test tests/dc_packet_storage_tests.cpp)
    endif()
    add_executable(dc-${kind}-wrapper-tests ${wrapper_test} ${outputs})
    if(kind MATCHES "^(packet_init|begin_frame|resident_chain|loading_init|packet_reserve|buffer_clear|packet_handle|packet_get)$")
      string(TOUPPER ${kind} resident_case)
      target_compile_definitions(dc-${kind}-wrapper-tests PRIVATE DC_${resident_case})
    endif()
    if(kind STREQUAL "item_state")
      target_compile_definitions(dc-${kind}-wrapper-tests PRIVATE DC_ITEM_COPY)
    endif()
    if(kind STREQUAL "shadow_fast")
      target_compile_definitions(dc-${kind}-wrapper-tests PRIVATE DC_SHADOW_FAST)
    elseif(kind STREQUAL "shadow_fast2")
      target_compile_definitions(dc-${kind}-wrapper-tests PRIVATE DC_SHADOW_FAST2)
    endif()
    if(kind STREQUAL "frame_draw" OR kind STREQUAL "frame_member")
      target_sources(dc-${kind}-wrapper-tests PRIVATE runtime/dc_native.cpp)
      target_compile_definitions(dc-${kind}-wrapper-tests PRIVATE DC_DEFAULT_TEXTURES="")
